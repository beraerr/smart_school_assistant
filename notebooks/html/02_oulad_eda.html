Veri¶

load_tables ile açılan tabloların boyutu basılır. studentVle bu fonksiyonun dışındadır; aynı tür dönüşümüyle parça parça sayılır. Hücre arşivin kayıtlı sağlama değerini ve satır birimini de basar. Satır birimi, bir öğrencinin bir modül sunumundaki kaydıdır.

def _bin_label(interval):
    return f"({float(interval.left):.2f}, {float(interval.right):.2f}]"


def rate_by_bin(frame, column, quantiles, missing_label):
    if quantiles is None:
        labels = frame[column].astype("int64").astype(str)
        categories = [str(value) for value in sorted(frame[column].astype("int64").unique())]
    else:
        observed = frame[column].notna()
        labels = pd.Series(pd.NA, index=frame.index, dtype="object")
        categories = []
        if observed.any():
            cut = pd.qcut(
                frame.loc[observed, column].astype("float64"),
                quantiles,
                duplicates="drop",
            )
            categories = [_bin_label(item) for item in cut.cat.categories]
            labels.loc[observed] = cut.map(_bin_label).to_numpy()
        if (~observed).any():
            categories.append(missing_label)
            labels.loc[~observed] = missing_label
        labels = labels.astype(str)
    work = pd.DataFrame({"bin": labels.to_numpy(), "at_risk": frame["at_risk"].to_numpy()})
    counts = work.groupby("bin", sort=False).size()
    rates = work.groupby("bin", sort=False)["at_risk"].mean()
    table = pd.DataFrame({"bin": categories})
    table["n"] = table["bin"].map(counts).fillna(0).astype(int)
    table["at_risk_rate"] = table["bin"].map(rates).astype(float)
    return table


def print_rate_table(table):

def plot_rate_bars(table, title, xlabel, ylabel):
    labels = table["bin"].astype(str).tolist()
    heights = table["at_risk_rate"].to_numpy(dtype=float)
    fig, ax = plt.subplots(figsize=(8.8, 4.6))
    ax.bar(labels, heights, color="#4c78a8", width=0.72)
    ax.set_ylim(0, 1.18)
    ax.set_xlabel(xlabel)
    ax.set_ylabel(ylabel)
    ax.set_title(title)
    ax.set_axisbelow(True)
    ax.yaxis.grid(True, linestyle=":", alpha=0.5)
    if any(len(label) > 4 for label in labels):
        ax.tick_params(axis="x", labelrotation=30)
        for tick in ax.get_xticklabels():
            tick.set_ha("right")
    for label, height, count in zip(labels, heights, table["n"].tolist()):
        if not np.isfinite(height):
            continue
        ax.text(label, height + 0.03, f"n={int(count)}", ha="center", va="bottom", fontsize=8)
    fig.tight_layout()
    plt.show()


def read_feature_catalog(path):
    rows = []
    for line in Path(path).read_text(encoding="utf-8").splitlines():
        if not line.startswith("|"):
            continue
        cells = [part.strip() for part in line.strip().strip("|").split("|")]
        if len(cells) != 4 or cells[0].startswith("-") or cells[0] == "sütun":
            continue
        rows.append(cells)
    return pd.DataFrame(rows, columns=["column", "definition", "scope", "school"])
